cmake_minimum_required(VERSION 3.16)
project(rsim_stm32 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/SetVelocity.srv" "srv/Stop.srv")
add_executable(stm32_node src/stm32_node.cpp)
target_include_directories(stm32_node PRIVATE include)
target_compile_options(stm32_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(stm32_node rclcpp nav_msgs sensor_msgs geometry_msgs std_msgs diagnostic_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(stm32_node "${typesupport}")
install(TARGETS stm32_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_protocol test/test_protocol.cpp)
  target_include_directories(test_protocol PRIVATE include)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
